    COMPANY_WEBSITE             "https://virgoaudiomasters.com"
    PLUGIN_MANUFACTURER_CODE    Vram
    PLUGIN_CODE                 VaDc
    FORMATS                     VST3 AU Standalone
    PRODUCT_NAME                "VA De-clip"

    NEEDS_MIDI_INPUT            FALSE
